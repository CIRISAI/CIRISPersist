New module `src/federation/edition2/voices.rs` (`pub mod voices;` in `edition2/mod.rs`). Path: `crate::federation::edition2::voices`.

```rust
use crate::federation::edition2::{Digest32, names::RowId};
use crate::federation::{Attestation, Error, FederationDirectory};

/// One held voice of a row: one signer's edition-2 statement over the row's
/// payload, as `ceg_statements` holds it. Verified at its door; NOT re-verified.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Voice {
    pub iss: String,          // CWT iss: the identity-ref (a person; a second device shares it)
    pub kid: Digest32,        // the signing key's thumbprint (protected header label 4)
    pub asserted_at_ms: u64,  // the payload's asserted_at (the same for every voice of one row id)
}

/// Every held voice of `row_id`, one per statement, in statement-id order.
/// Reads `FederationDirectory::edition2_voices` (ceg_statements, both backends).
/// A held statement that does not decode as a row statement of exactly `row_id`
/// is skipped (never counted), not an error.
pub async fn voices_of_row<F>(dir: &F, row_id: &RowId) -> Result<Vec<Voice>, Error>
where F: FederationDirectory + ?Sized;

/// The edition-2 row id of a statement-carrying row (`attestation_id` = hex row id,
/// envelope carries the statement). `None` for an earlier-form row.
pub fn carried_row_id(row: &Attestation) -> Option<RowId>;

/// The DISTINCT `iss` among `row_id`'s voices that are in `roster`
/// (roster = identity-refs / key ids as the typed group rows hold them).
pub async fn roster_voices<F>(dir: &F, row_id: &RowId, roster: &[String])
    -> Result<std::collections::BTreeSet<String>, Error>
where F: FederationDirectory + ?Sized;

/// The ONE switch every converted gate calls in place of
/// `reverse_quorum::count_distinct_roster_scrubs(dir, &row.attestation_envelope, &row.scrubs(), roster)`:
///  - row carries a statement      -> roster_voices(dir, carried_row_id, roster)
///  - earlier-form row, doors open -> count_distinct_roster_scrubs(..) (unchanged; F deletes)
///  - earlier-form row, doors shut -> empty set (counts 0)
pub async fn counted_roster_signers<F>(dir: &F, row: &Attestation, roster: &[String])
    -> Result<std::collections::BTreeSet<String>, Error>
where F: FederationDirectory + ?Sized;

/// v2 — the voice of the statement the row carries (its admitting voice).
pub fn carried_voice(row: &Attestation) -> Option<Voice>;

/// v2 — every distinct `iss` of a statement-carrying row (held voices + its carried voice),
/// no roster filter. Empty for an earlier-form row.
pub async fn row_voice_issuers<F>(dir: &F, row: &Attestation)
    -> Result<std::collections::BTreeSet<String>, Error>
where F: FederationDirectory + ?Sized;
```
